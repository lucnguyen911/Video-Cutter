704:     last_boundary_time = 0.0
705: 
706:     for split_time in normalized_splits:
707:         nominal_left = round(split_time - smooth_seconds, 3)
708:         nominal_right = round(split_time + smooth_seconds, 3)
709: 
710:         adjusted_left = nominal_left
711:         adjusted_right = nominal_right
712: 
713:         # 4.1. Kiểm tra phía trái
714:         while True:
715:             idx = bisect_left(normalized_scenes, adjusted_left)
716:             if idx > 0:
717:                 prev_scene = normalized_scenes[idx - 1]
718:                 dist = adjusted_left - prev_scene
719:                 ext = nominal_left - prev_scene
720:                 if dist <= short_fragment_threshold and ext <= max_fragment_extension:
721:                     if prev_scene > 0:
722:                         adjusted_left = prev_scene
723:                     else:
724:                         break
725:                 else:
726:                     break
727:             else:
728:                 break
729: 
730:         # 4.2. Kiểm tra phía phải
731:         while True:
732:             idx = bisect_right(normalized_scenes, adjusted_right)
733:             if idx < len(normalized_scenes):
734:                 next_scene = normalized_scenes[idx]
735:                 dist = next_scene - adjusted_right
736:                 ext = next_scene - nominal_right
737:                 if dist <= short_fragment_threshold and ext <= max_fragment_extension:
738:                     if next_scene < duration:
739:                         adjusted_right = next_scene
740:                     else:
741:                         break
742:                 else:
743:                     break
744:             else:
745:                 break

748:         adjusted_right = round(adjusted_right, 3)
749:
